# updated command versions with filtering
libcamera-vid -v 0 -t 0 -b 5000000 --width 1920 --height 1080 --framerate 30 --codec h264 --inline --output - | \
ffmpeg -i - -f alsa -thread_queue_size 1024 -ac 1 -ar 48000 -i plughw:1,0 \
-filter_complex \
"[1:a]aformat=sample_fmts=fltp:channel_layouts=mono,asplit=2[a1][a2]; \
 [a1]equalizer=f=300:width_type=h:width=600:g=-45, \
      equalizer=f=1000:width_type=h:width=2000:g=-13, \
      firequalizer=gain='if(gt(f,300),0,-INF)', \
      afftdn=nf=-50, \
      volume=10dB[a1out]; \
 [a2]equalizer=f=300:width_type=h:width=600:g=-25, \
      equalizer=f=1000:width_type=h:width=2000:g=-6, \
      firequalizer=gain='if(gt(f,200),0,-INF)', \
      afftdn=nf=-30, \
      volume=10dB[a2out]" \
-vcodec copy -acodec libmp3lame -b:a 48k \
-map 0:v -map "[a1out]" -map "[a2out]" \
-f rtsp rtsp://localhost:8554/unicast \
-hide_banner -nostats
